library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

entity rulet is
    generic(
        WIN_MODE : integer := 50;
        DEBUG_MODE : boolean := true
    );
    Port (
        clk        : in  std_logic;
        ini        : in std_logic;
        reset      : in  std_logic;
        sw         : in  std_logic_vector(3 downto 0);
        acep       : in  std_logic;
        wol        : out std_logic;
        leds      : out std_logic_vector(3 downto 0);
        update     : out std_logic
    );
end rulet;
architecture Behavioral of rulet is
    type state_type is (ESP,IDLE, RESULTADO, DECISION, PAGO,RES);
    signal state : state_type:= ESP;
    signal apu_reg   : std_logic := '0';
    signal resu_reg  : std_logic := '0';
    signal acep_prev : std_logic := '0';
    signal aleatorio : std_logic;
    signal lfsr : std_logic_vector(3 downto 0) := "1010";
    signal direc_fpga : std_logic_vector(3 downto 0);
    signal wol_reg: std_logic;
    signal cambio : std_logic;
begin
process(clk)
begin
    if rising_edge(clk) then
        lfsr <= lfsr(2 downto 0) & (lfsr(3) xor lfsr(2));--Producci n de vector de 4 bits seudoaleatorio
    end if;
end process;
process(lfsr)
begin
    case WIN_MODE is--Seg n win mode cambia la generaci n seudoaleatoria
        when 25 =>
            case lfsr(2 downto 1) is
                when "00" =>
                    aleatorio <= '1';
                when others =>
                    aleatorio <= '0';
            end case;
        when 75 =>
            case lfsr(2 downto 1) is
                when "11" =>
                    aleatorio <= '0';
                when others =>
                    aleatorio <= '1';
            end case;
        when others => -- 50%
            case lfsr(2 downto 1) is
                when "00" => aleatorio <= '0';
                when "01" => aleatorio <= '1';
                when "10" => aleatorio <= '0';
                when others => aleatorio <= '1';
            end case;
    end case;
end process;
process(clk, reset)
    variable apu_var   : std_logic;--La apuesta se puede cambiar despues de generar el resultado
begin
    if reset = '1' then
        state     <= ESP;
    elsif rising_edge(clk) then
        acep_prev <= acep;
        update <= '0';--Se al de salida siempre 0 salvo durante 1 clk
        case state is
            when ESP =>--Se fuerza todo a 0
                apu_reg   <= '0';
                resu_reg  <= '0';
                acep_prev <= '0';
                wol       <= '0';
                update    <= '0';
                cambio <='0';
                wol_reg <= '0';           
                if ini='1' then--Se al de inicio que envia la FSM
                    state<=IDLE;
                end if;
            when IDLE =>
                if acep = '1' and acep_prev = '0' then--Al presionar el boton se toma que opci n se eligio
                    apu_reg <= sw(0);
                    state   <= RESULTADO;
                end if;
            when RESULTADO =>--Obtiene el resultado de la ruleta, notar que intencionalmente se muestra este resultado en los leds para poder determinar si funciona bien la aleatoriedad
                resu_reg <= aleatorio;
                state <= DECISION;
            when DECISION =>--Estado donde usuario decide cambiar la apuesta o no, 1=Cambio de apuesta
                if acep = '1' and acep_prev = '0' then--Cuando se apreta el boton se asigna la variable al registro, se determina si se cambia o no la variable 
                    apu_var := apu_reg;
                    if sw(0) = '1' then--Cambio de apuesta
                        apu_var := not apu_var;
                        cambio <='1';
                    end if;
                    apu_reg <= apu_var;--Asignaci n del cambio, todo dentro de 1 clk
                    state <= PAGO;
                end if;
           when PAGO =>--Se determina si gana o pierde
                if apu_reg = resu_reg then
                    wol_reg <= '1';
                else
                    wol_reg <= '0';
                end if;        
                state <= RES;
           when RES =>
                if (acep = '1' and acep_prev = '0') then--Solo en ese clk se activa el proceso, al apretar el boton se envia las se ales de aviso a la FSM
                    wol<=wol_reg;
                    update<='1';
                    state <= ESP;  --Vuelta al estado de espera
               end if;              
        end case;
    end if;
end process;
process(state, resu_reg, wol_reg)
begin
    if DEBUG_MODE then
        leds <= resu_reg & cambio & '0' & wol_reg;--En modo debug permite ver el resultado de la ruleta
    else
        leds <= cambio & '0' & apu_reg & wol_reg;--En modo estandar solo muestra si hubo cambio en la apuesta, la apuesta y si gan 
    end if;
end process;
end Behavioral;
