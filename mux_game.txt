library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity game_selector is --Toma las señales de salida wol y update/validacion de cada juego y elije cual enviar a la FSM
    Port (
        clk    : in std_logic; 
        rst    : in std_logic;
        update_1 : in std_logic;
        wol_1  : in std_logic;
        update_2 : in std_logic;
        wol_2  : in std_logic;
        update_3 : in std_logic;
        wol_3  : in std_logic;
        update_out : out std_logic;
        wol_out  : out std_logic
    );
end game_selector;
architecture Behavioral of game_selector is
begin
    process(clk, rst)
    begin
        if rst = '1' then
            update_out <= '0';
            wol_out  <= '0';    
        elsif rising_edge(clk) then--Forzar que dure 1 clk distinto de o las salidas
            update_out <= '0';
            wol_out  <= '0';
            case std_logic_vector'(update_3 & update_2 & update_1) is--Según de donde viene el bit de validación se asigna la salida
                when "001" =>
                    update_out <= update_1;
                    wol_out  <= wol_1;
                when "010" =>
                    update_out <= update_2;
                    wol_out  <= wol_2;                 
                when "100" =>
                    update_out <= update_3;
                    wol_out  <= wol_3;
                when others => 
                    update_out <= '0';
                    wol_out  <= '0';
            end case;
        end if;
    end process;
end Behavioral;
